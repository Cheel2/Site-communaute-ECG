import { type NextRequest, NextResponse } from "next/server";
import { createServerClient, type CookieOptions } from "@supabase/ssr";

export async function middleware(request: NextRequest) {
  // 1. Vérifier si la route est publique (admin/login)
  const pathname = request.nextUrl.pathname;

  // Exclure /admin/login et /admin/mot-de-passe-reinitialiser de la protection
  if (pathname === "/admin/login" || pathname === "/admin/mot-de-passe-reinitialiser") {
    return NextResponse.next();
  }

  // 2. Vérifier les variables d'environnement
  const supabaseUrl = process.env.NEXT_PUBLIC_SUPABASE_URL;
  const supabaseAnonKey = process.env.NEXT_PUBLIC_SUPABASE_ANON_KEY;

  if (!supabaseUrl || !supabaseAnonKey) {
    return NextResponse.next();
  }

  // 3. Créer le client Supabase
  let response = NextResponse.next({ request });

  const supabase = createServerClient(supabaseUrl, supabaseAnonKey, {
    cookies: {
      getAll() {
        return request.cookies.getAll();
      },
      setAll(cookiesToSet: { name: string; value: string; options: CookieOptions }[]) {
        cookiesToSet.forEach(({ name, value }) => {
          request.cookies.set(name, value);
        });

        response = NextResponse.next({ request });

        cookiesToSet.forEach(({ name, value, options }) => {
          response.cookies.set(name, value, options);
        });
      },
    },
  });

  // 4. Vérifier l'authentification
  const {
    data: { user },
  } = await supabase.auth.getUser();

  // 5. Si non authentifié ET route admin (sauf login déjà exclu) → rediriger
  if (!user) {
    const redirectUrl = request.nextUrl.clone();
    const redirectedFrom = `${request.nextUrl.pathname}${request.nextUrl.search}`;

    redirectUrl.pathname = "/admin/login";
    redirectUrl.search = "";
    redirectUrl.searchParams.set("redirectedFrom", redirectedFrom);

    return NextResponse.redirect(redirectUrl);
  }

  return response;
}

export const config = {
  matcher: ["/admin/:path*"],
};
